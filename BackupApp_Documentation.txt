## 🐍 Paso 1: Crear la Aplicación Python

### Estructura básica del script (`backupapp.py`):

```python
#!/usr/bin/env python3
"""
BackupApp - Aplicación para crear backups en formato ZIP
Autor: [Tu Nombre]
Fecha: [Fecha]
"""

import sys
import os
from PyQt6.QtWidgets import (QApplication, QMainWindow, QVBoxLayout, 
                             QPushButton, QLabel, QFileDialog, QMessageBox)

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.initUI()
    
    def initUI(self):
        self.setWindowTitle("BackupApp")
        self.setGeometry(100, 100, 400, 200)
        
        # Widgets principales
        self.label = QLabel("Bienvenido a BackupApp")
        self.button = QPushButton("Seleccionar Carpeta")
        self.button.clicked.connect(self.select_folder)
        
        # Layout
        layout = QVBoxLayout()
        layout.addWidget(self.label)
        layout.addWidget(self.button)
        
        container = QWidget()
        container.setLayout(layout)
        self.setCentralWidget(container)
    
    def select_folder(self):
        folder = QFileDialog.getExistingDirectory(self, "Seleccionar Carpeta")
        if folder:
            self.label.setText(f"Carpeta seleccionada: {folder}")

def main():
    app = QApplication(sys.argv)
    window = MainWindow()
    window.show()
    sys.exit(app.exec())

if __name__ == "__main__":
    main()
```

### Consideraciones importantes:
- ✅ Usar rutas relativas, no absolutas
- ✅ Incluir manejo de errores
- ✅ No asumir dependencias del sistema
- ✅ Usar `sys.exit(app.exec())` para cierre limpio

## 🛠️ Paso 3: Script de Construcción del AppImage

### Script completo (`build-appimage.sh`):

```bash
#!/bin/bash

# =============================================
# SCRIPT DE CONSTRUCCIÓN DE APPIMAGE
# BackupApp - Versión 1.0
# =============================================

# Configuración
APP_NAME="BackupApp"
APPDIR="${APP_NAME}.AppDir"
APPIMAGE_NAME="${APP_NAME}-x86_64.AppImage"

# Colores para output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Funciones de logging
log_info() {
    echo -e "${BLUE}[INFO]${NC} $1"
}

log_success() {
    echo -e "${GREEN}[SUCCESS]${NC} $1"
}

log_warning() {
    echo -e "${YELLOW}[WARNING]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# Verificar requisitos
check_requirements() {
    log_info "Verificando requisitos del sistema..."
    
    # Verificar Python
    if ! command -v python3 &> /dev/null; then
        log_error "Python3 no está instalado"
        exit 1
    fi
    
    # Verificar pip
    if ! command -v pip &> /dev/null; then
        log_error "pip no está instalado"
        exit 1
    fi
    
    # Verificar archivos necesarios
    if [ ! -f "backupapp.py" ]; then
        log_error "No se encuentra backupapp.py"
        exit 1
    fi
    
    if [ ! -f "icon.png" ]; then
        log_warning "No se encuentra icon.png, se usará icono por defecto"
    fi
    
    log_success "Requisitos verificados correctamente"
}

# Limpiar builds anteriores
clean_previous_builds() {
    log_info "Limpiando builds anteriores..."
    
    rm -rf "$APPDIR"
    rm -rf build
    rm -rf dist
    rm -f "$APPIMAGE_NAME"
    
    log_success "Limpieza completada"
}

# Preparar entorno virtual
setup_venv() {
    log_info "Preparando entorno virtual..."
    
    if [ ! -d "venv" ]; then
        log_info "Creando entorno virtual..."
        python3 -m venv venv
    fi
    
    # Activar entorno virtual
    source venv/bin/activate
    
    # Instalar/actualizar dependencias
    log_info "Instalando dependencias..."
    pip install --upgrade pip
    pip install PyQt6 pyinstaller
    
    log_success "Entorno virtual preparado"
}

# Crear ejecutable con PyInstaller
create_executable() {
    log_info "Creando ejecutable con PyInstaller..."
    
    # Variables de entorno para compatibilidad
    export CFLAGS="-march=x86-64 -mtune=generic"
    export CXXFLAGS="-march=x86-64 -mtune=generic"
    
    pyinstaller --onefile --windowed --name="$APP_NAME" \
        --hidden-import='PyQt6.QtCore' \
        --hidden-import='PyQt6.QtGui' \
        --hidden-import='PyQt6.QtWidgets' \
        --add-data "icon.png:." \
        backupapp.py
    
    # Verificar que el ejecutable se creó
    if [ ! -f "dist/$APP_NAME" ]; then
        log_error "Falló la creación del ejecutable"
        exit 1
    fi
    
    log_success "Ejecutable creado: dist/$APP_NAME"
}

# Crear estructura AppDir
create_appdir() {
    log_info "Creando estructura AppDir..."
    
    # Crear directorios necesarios
    mkdir -p "$APPDIR"/usr/bin
    mkdir -p "$APPDIR"/usr/lib
    mkdir -p "$APPDIR"/usr/share/icons/hicolor/256x256/apps
    mkdir -p "$APPDIR"/usr/share/applications
    
    # Copiar ejecutable
    cp "dist/$APP_NAME" "$APPDIR/usr/bin/"
    
    # Copiar icono si existe
    if [ -f "icon.png" ]; then
        cp icon.png "$APPDIR/usr/share/icons/hicolor/256x256/apps/backupapp.png"
    fi
    
    log_success "Estructura AppDir creada"
}

# Crear archivo .desktop
create_desktop_file() {
    log_info "Creando archivo .desktop..."
    
    cat > "$APPDIR/$APP_NAME.desktop" << EOF
[Desktop Entry]
Name=$APP_NAME
Comment=Application to create ZIP backups
Exec=AppRun
Icon=backupapp
Type=Application
Categories=Utility;
Terminal=false
StartupWMClass=BackupApp
X-AppImage-Name=$APP_NAME
X-AppImage-Version=1.0
EOF

    log_success "Archivo .desktop creado"
}

# Crear AppRun
create_apprun() {
    log_info "Creando AppRun..."
    
    cat > "$APPDIR/AppRun" << 'EOF'
#!/bin/bash
HERE="$(dirname "$(readlink -f "${0}")")"
export PATH="$HERE/usr/bin:$PATH"
export LD_LIBRARY_PATH="$HERE/usr/lib:$LD_LIBRARY_PATH"
cd "$HERE"
exec "./usr/bin/BackupApp" "$@"
EOF

    chmod +x "$APPDIR/AppRun"
    log_success "AppRun creado y hecho ejecutable"
}

# Crear enlaces simbólicos
create_symlinks() {
    log_info "Creando enlaces simbólicos..."
    
    cd "$APPDIR"
    
    # Enlace del icono
    if [ -f "usr/share/icons/hicolor/256x256/apps/backupapp.png" ]; then
        ln -sf usr/share/icons/hicolor/256x256/apps/backupapp.png ./
    fi
    
    cd ..
    
    log_success "Enlaces simbólicos creados"
}

# Generar AppImage
generate_appimage() {
    log_info "Generando AppImage..."
    
    # Verificar si appimagetool está disponible
    if command -v appimagetool &> /dev/null; then
        log_info "Usando appimagetool del sistema..."
        appimagetool "$APPDIR"
    else
        log_warning "appimagetool no encontrado en PATH"
        log_info "Intentando descargar appimagetool..."
        
        # Descargar appimagetool
        wget -q https://github.com/AppImage/AppImageKit/releases/download/continuous/appimagetool-x86_64.AppImage
        chmod +x appimagetool-x86_64.AppImage
        ./appimagetool-x86_64.AppImage "$APPDIR"
        rm -f appimagetool-x86_64.AppImage
    fi
    
    # Verificar que el AppImage se creó
    if [ -f "$APPIMAGE_NAME" ]; then
        log_success "AppImage generado: $APPIMAGE_NAME"
    else
        log_error "Falló la generación del AppImage"
        exit 1
    fi
}

# Función principal
main() {
    echo "============================================="
    echo "   CONSTRUYENDO APPIMAGE PARA BACKUPAPP"
    echo "============================================="
    
    # Ejecutar pasos en orden
    check_requirements
    clean_previous_builds
    setup_venv
    create_executable
    create_appdir
    create_desktop_file
    create_apprun
    create_symlinks
    generate_appimage
    
    echo ""
    echo "============================================="
    log_success "CONSTRUCCIÓN COMPLETADA EXITOSAMENTE"
    echo "============================================="
    echo ""
    echo "Archivo generado: $(pwd)/$APPIMAGE_NAME"
    echo ""
    echo "Para ejecutar: ./$APPIMAGE_NAME"
    echo "Para distribuir: Copia el archivo .AppImage a otros sistemas"
}

# Ejecutar función principal
main "$@"
```

## 🚀 Paso 4: Proceso de Construcción

### 1. Hacer ejecutable el script:
```bash
chmod +x build-appimage.sh
```

### 2. Ejecutar construcción:
```bash
./build-appimage.sh
```

### 3. El proceso automático ejecutará:
- ✅ Verificación de requisitos
- ✅ Limpieza de builds anteriores  
- ✅ Configuración del entorno virtual
- ✅ Creación del ejecutable con PyInstaller
- ✅ Construcción de la estructura AppDir
- ✅ Generación del AppImage final

## 🔧 Solución de Problemas Comunes

### 1. Error: "No se encuentra backupapp.py"
**Solución**: Ejecutar el script desde la carpeta correcta

### 2. Error: "PyQt6 no está instalado"
**Solución**: 
```bash
source venv/bin/activate
pip install PyQt6
```

### 3. Error: "Illegal Instruction" en CPUs antiguas
**Solución**: Modificar las flags de compilación en el script:
```bash
export CFLAGS="-march=x86-64 -mtune=generic -mno-avx -mno-avx2"
export CXXFLAGS="-march=x86-64 -mtune=generic -mno-avx -mno-avx2"
```

### 4. Error: "appimagetool no encontrado"
**Solución**: El script lo descarga automáticamente

### 5. La aplicación no muestra el icono
**Solución**: Verificar que `icon.png` existe y es de 256x256px

## 💡 Consejos para Mantenimiento

### Actualizar la aplicación:
1. Modificar `backupapp.py`
2. Ejecutar `./build-appimage.sh` nuevamente
3. El script se encarga de todo automáticamente

### Agregar nuevas dependencias:
```bash
source venv/bin/activate
pip install nueva-dependencia
./build-appimage.sh
```

### Cambiar el icono:
- Reemplazar `icon.png`
- Mantener el mismo nombre y formato
- Ejecutar `./build-appimage.sh`
